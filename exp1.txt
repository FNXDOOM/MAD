MainActivity.kt


package com.example.calculator

import android.os.Bundle
import android.widget.Button
import android.widget.EditText
import android.widget.TextView
import androidx.activity.enableEdgeToEdge
import androidx.appcompat.app.AppCompatActivity

class MainActivity : AppCompatActivity() {

    lateinit var btn0: Button
    lateinit var btn1: Button
    lateinit var btn2: Button
    lateinit var btn3: Button
    lateinit var btn4: Button
    lateinit var btn5: Button
    lateinit var btn6: Button
    lateinit var btn7: Button
    lateinit var btn8: Button
    lateinit var btn9: Button

    lateinit var btnPlus: Button
    lateinit var btnSub: Button
    lateinit var btnMul: Button
    lateinit var btnDiv: Button
    lateinit var btnClr: Button
    lateinit var btnEq: Button

    lateinit var TView: EditText
    lateinit var Display: TextView

    var firstValue = 0.0
    var secondValue = 0.0
    var result = 0.0
    var op = ""

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContentView(R.layout.activity_main)

        btn0 = findViewById(R.id.btn0)
        btn1 = findViewById(R.id.btn1)
        btn2 = findViewById(R.id.btn2)
        btn3 = findViewById(R.id.btn3)
        btn4 = findViewById(R.id.btn4)
        btn5 = findViewById(R.id.btn5)
        btn6 = findViewById(R.id.btn6)
        btn7 = findViewById(R.id.btn7)
        btn8 = findViewById(R.id.btn8)
        btn9 = findViewById(R.id.btn9)

        btnPlus = findViewById(R.id.btnPlus)
        btnSub = findViewById(R.id.btnSub)
        btnMul = findViewById(R.id.btnMul)
        btnDiv = findViewById(R.id.btnDiv)
        btnClr = findViewById(R.id.btnCLR)
        btnEq = findViewById(R.id.btnEq)

        TView = findViewById(R.id.Edt1)
        Display = findViewById(R.id.Display)

        // Number buttons
        btn0.setOnClickListener { TView.append("0") }
        btn1.setOnClickListener { TView.append("1") }
        btn2.setOnClickListener { TView.append("2") }
        btn3.setOnClickListener { TView.append("3") }
        btn4.setOnClickListener { TView.append("4") }
        btn5.setOnClickListener { TView.append("5") }
        btn6.setOnClickListener { TView.append("6") }
        btn7.setOnClickListener { TView.append("7") }
        btn8.setOnClickListener { TView.append("8") }
        btn9.setOnClickListener { TView.append("9") }

        // Operator buttons
        btnPlus.setOnClickListener {
            if (TView.text.isNotEmpty()) {
                firstValue = TView.text.toString().toDouble()
                op = "+"
                TView.text.clear()
            }
        }

        btnSub.setOnClickListener {
            if (TView.text.isNotEmpty()) {
                firstValue = TView.text.toString().toDouble()
                op = "-"
                TView.text.clear()
            }
        }

        btnMul.setOnClickListener {
            if (TView.text.isNotEmpty()) {
                firstValue = TView.text.toString().toDouble()
                op = "*"
                TView.text.clear()
            }
        }

        btnDiv.setOnClickListener {
            if (TView.text.isNotEmpty()) {
                firstValue = TView.text.toString().toDouble()
                op = "/"
                TView.text.clear()
            }
        }

        // Clear button
        btnClr.setOnClickListener {
            TView.text.clear()
            Display.text = ""
            firstValue = 0.0
            secondValue = 0.0
            result = 0.0
            op = ""
        }

        // Equals button
        btnEq.setOnClickListener {

            if (TView.text.isEmpty()) {
                return@setOnClickListener
            }

            secondValue = TView.text.toString().toDouble()

            result = when (op) {
                "+" -> firstValue + secondValue
                "-" -> firstValue - secondValue
                "*" -> firstValue * secondValue

                "/" -> {
                    if (secondValue == 0.0) {
                        Display.text = "Error: Div by 0"
                        return@setOnClickListener
                    }
                    firstValue / secondValue
                }

                else -> 0.0
            }

            Display.text = result.toString()
            TView.setText(result.toString())
        }
    }
}
